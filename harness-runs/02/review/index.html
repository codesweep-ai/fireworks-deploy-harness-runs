"""Reusable real-HTTP plumbing for task-tracker tests (Python stdlib only).

Base URL is http://127.0.0.1:8080 (override with TT_BASE_URL). Each test case
gets its own cookie jar, i.e. its own session. Redirects are NOT followed, so
tests can assert on 302/Location themselves. 4xx/5xx responses are returned as
response objects instead of raising.
"""
import http.cookiejar
import os
import unittest
import urllib.error
import urllib.parse
import urllib.request

BASE_URL = os.environ.get("TT_BASE_URL", "http://127.0.0.1:8080").rstrip("/")


class _NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return None


class HttpCase(unittest.TestCase):
    """Base class: session-cookie HTTP client plus login/action helpers."""

    def setUp(self):
        self.jar = http.cookiejar.CookieJar()
        self.opener = urllib.request.build_opener(
            urllib.request.ProxyHandler({}),
            urllib.request.HTTPCookieProcessor(self.jar),
            _NoRedirect(),
        )

    def request(self, method, path, params=None):
        data = None if params is None else urllib.parse.urlencode(params).encode()
        req = urllib.request.Request(BASE_URL + path, data=data, method=method)
        try:
            return self.opener.open(req, timeout=10)
        except urllib.error.HTTPError as exc:  # 4xx/5xx are answers, not crashes
            return exc

    def login(self, username, password):
        """POST /login. Success: 302 to /tasks + JSESSIONID cookie.
        Failure: 200, login page with 'Invalid credentials'."""
        return self.request("POST", "/login", {"username": username, "password": password})

    def task_action(self, task_id, action, **extra):
        """POST /tasks/action as the currently logged-in user."""
        params = {"id": str(task_id), "action": action}
        params.update(extra)
        return self.request("POST", "/tasks/action", params)

    def session_cookie(self):
        return next((c for c in self.jar if c.name == "JSESSIONID"), None)
